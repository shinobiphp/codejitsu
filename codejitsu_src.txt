    41	        $codex = $this->kernelInstance->codex;
    42	
    43	        // 2. Dispatch through the pipeline and execute Codex command
    44	        $result = $this->pipeline->send($intent, function (Intent $i) use ($codex) {
    45	            if (!$codex->has($i->action)) {
    46	                fwrite(STDERR, "Unknown command action: {$i->action}\n");
    47	                return 1;
    48	            }
    49	
    50	            $handler = $codex->get($i->action);
    51	            return (int) $handler($i);
    52	        });
    53	
    54	        return is_int($result) ? $result : 0;
    55	    }

    23	    public Kernel $kernel {
    24	        get => $this->kernelInstance;

    27	    public function __construct(
    28	        private readonly Kernel $kernelInstance,

    30	        private readonly int $port = 9501
    31	    ) {

    33	        $this->server = new Server($this->host, $this->port);
    34	    }
    35	
    36	    public function use(Middleware|Closure ...$middleware): self
    37	    {
    38	        $this->pipeline->pipe(...$middleware);
    39	        return $this;
    40	    }
    41	
    42	    public function run(mixed ...$args): void
    43	    {
    44	        $settings = $args[0] ?? [];
    45	        if (!empty($settings) && is_array($settings)) {
    46	            $this->server->set($settings);
    47	        }
    48	
    49	        $this->server->on('request', function (Request $request, Response $response): void {
    50	            // 1. Translate OpenSwoole Request into a typed HttpIntent
    51	            $intent = HttpTranslator::fromSwoole($request);
    52	            $codex = $this->kernelInstance->codex;
    53	
    54	            // 2. Dispatch through pipeline per coroutine request
    55	            $this->pipeline->send($intent, function (Intent $i) use ($codex, $response) {
    56	                if ($codex->has($i->action)) {
    57	                    $handler = $codex->get($i->action);
    58	                    $result = $handler($i);
    59	
    60	                    $response->end(is_array($result) ? json_encode($result) : (string) $result);
    61	                    return $result;
    62	                }
    63	
    64	                $response->status(404);
    65	                $response->end('Not Found');
    66	                return null;
    67	            });
    68	        });
    69	
    70	        $this->server->start();
    71	    }
    72	}

     2

     5	namespace Codejitsu\Codecs;
     6	
     7	use Codejitsu\Contracts\Codec as CodecContract;
     8	use Codejitsu\Contracts\Crypto\Sealer as SealerContract;
     9	use Codejitsu\Contracts\Crypto\Signer as SignerContract;

    12	use Codejitsu\Enums\Crypto\SignatureAlgorithms;
    13	use RuntimeException;
    14	
    15	abstract class Codec implements CodecContract
    16	{
    17	    protected const FORMAT = 'neon';
    18	
    19	    protected const SEALER = 'aes-256-gcm';
    20	
    21	    protected const SIGNER = 'hmac-sha256';
    22	
    23	    private ?SealerContract $sealerInstance = null;
    24	
    25	    private ?SignerContract $signerInstance = null;
    26	
    27	    public ?SealerContract $sealer {

    30	    }
    31	
    32	    public ?SignerContract $signer {
    33	        get => $this->signerInstance ??= $this->createSigner();
    34	        set (?SignerContract $value) => $this->signerInstance = $value;
    35	    }
    36	
    37	    public function format(): CodecFormat
    38	    {
    39	        return CodecFormat::tryFrom(static::FORMAT)
    40	            ?? throw new RuntimeException(
    41	                sprintf(
    42	                    'Invalid codec format [%s] defined in [%s].',
    43	                    static::FORMAT,
    44	                    static::class,
    45	                ),
    46	            );
    47	    }
    48	
    49	    public function setSealer(
    50	        SealerContract $sealer,
    51	    ): static {
    52	        $this->sealer = $sealer;
    53	
    54	        return $this;
    55	    }
    56	
    57	    public function setSigner(
    58	        SignerContract $signer,
    59	    ): static {
    60	        $this->signer = $signer;
    61	
    62	        return $this;
    63	    }
    64	
    65	    public function isSealed(
    66	        string $filePathOrPayload,
    67	    ): bool {
    68	        $sealer = $this->sealer;
    69	
    70	        if ($sealer === null) {
    71	            return false;
    72	        }
    73	
    74	        return $sealer->isSealed(
    75	            $this->resolvePayload($filePathOrPayload),
    76	        );
    77	    }
    78	
    79	    public function inspect(
    80	        string $filePathOrPayload,
    81	        ?string $key = null,
    82	    ): array {
    83	        $payload = $this->resolvePayload(
    84	            $filePathOrPayload,
    85	        );
    86	
    87	        $sealer = $this->sealer;
    88	
    89	        if (
    90	            $sealer !== null
    91	            && $sealer->isSealed($payload)
    92	        ) {
    93	            if ($key === null) {
    94	                throw new RuntimeException(
    95	                    'A decryption key is required to inspect a sealed payload.',
    96	                );
    97	            }
    98	
    99	            $payload = $sealer->unseal(
   100	                $payload,
   101	                $key,
   102	            );
   103	        }
   104	
   105	        return $this->decode(
   106	            $payload,
   107	            $key,
   108	        );
   109	    }
   110	
   111	    abstract public function encode(
   112	        array $data,
   113	    ): string;
   114	
   115	    abstract public function decode(
   116	        string $payload,
   117	        ?string $key = null,
   118	    ): array;
   119	
   120	    protected function createSealer(): ?SealerContract
   121	    {
   122	        if (static::SEALER === null) {
   123	            return null;
   124	        } else {
   125	            $algorithm = EncryptionAlgorithms::tryFrom(
   126	                static::SEALER,
   127	            );
   128	        }
   129	
   130	        if ($algorithm === null) {
   131	            throw new RuntimeException(
   132	                sprintf(
   133	                    'Invalid sealer algorithm [%s] in [%s].',
   134	                    static::SEALER,
   135	                    static::class,
   136	                ),
   137	            );
   138	        }
   139	
   140	        $factory = $algorithm->to('$sealer');
   141	
   142	        if ($factory instanceof \Closure) {
   143	            $sealer = $factory();
   144	
   145	            if (!$sealer instanceof SealerContract) {
   146	                throw new RuntimeException(
   147	                    sprintf(
   148	                        'Sealer factory for [%s] did not return a SealerContract.',
   149	                        $algorithm->value,
   150	                    ),
   151	                );
   152	            }
   153	
   154	            return $sealer;
   155	        }
   156	
   157	        if (
   158	            is_string($factory)
   159	            && class_exists($factory)
   160	        ) {
   161	            $sealer = new $factory();
   162	
   163	            if (!$sealer instanceof SealerContract) {
   164	                throw new RuntimeException(
   165	                    sprintf(
   166	                        'Sealer class [%s] does not implement SealerContract.',
   167	                        $factory,
   168	                    ),
   169	                );
   170	            }
   171	
   172	            return $sealer;
   173	        }
   174	
   175	        throw new RuntimeException(
   176	            sprintf(
   177	                'No valid sealer factory configured for [%s].',
   178	                $algorithm->value,
   179	            ),
   180	        );
   181	    }

   219	
   220	        if (
   221	            is_string($factory)

   225	
   226	            if (!$signer instanceof SignerContract) {

   246	    protected function resolvePayload(

   259	                sprintf(
   260	                    'Unable to read codec payload file [%s].',
   261	                    $filePathOrPayload,
   262	                ),
   263	            );
   264	        }
   265	
   266	        return $payload;
   267	    }

    12	final class OpenSSL implements SealerContract
    13	{

    15	
    16	    protected const TAG_LENGTH = 16;
    17	
    18	    protected const MINIMUM_PAYLOAD_LENGTH = 1;

    20	    public function __construct(
    21	        protected string $cipher = self::CIPHER,
    22	    ) {

    24	            openssl_cipher_iv_length($this->cipher) === false

    41	    {
    42	        $raw = base64_decode($payload, true);
    43	
    44	        if ($raw === false) {
    45	            return false;
    46	        }
    47	
    48	        $ivLength = openssl_cipher_iv_length(
    49	            $this->cipher,
    50	        );
    51	
    52	        if ($ivLength === false) {
    53	            return false;
    54	        }
    55	
    56	        return strlen($raw) >= (
    57	            $ivLength

    70	            $this->cipher,
    71	        );
    72	
    73	        if ($ivLength === false) {
    74	            throw new RuntimeException(
    75	                sprintf(

    78	                ),
    79	            );
    80	        }
    81

    83	        $tag = '';
    84	
    85	        $encrypted = openssl_encrypt(
    86	            $payload,
    87	            $this->cipher,
    88	            $key,
    89	            OPENSSL_RAW_DATA,
    90	            $iv,
    91	            $tag,
    92	            '',
    93	            self::TAG_LENGTH,
    94	        );
    95	
    96	        if ($encrypted === false) {
    97	            throw new RuntimeException(
    98	                'Failed to seal payload using OpenSSL.',
    99	            );
   100	        }

   104	            . $tag
   105	            . $encrypted,
   106	        );
   107	    }

   113	        $key = $this->normalizeKey($decryptionKey);
   114

   116	            $sealedPayload,
   117	            true,
   118	        );
   119	
   120	        if ($raw === false) {
   121	            throw new RuntimeException(
   122	                'Invalid base64 sealed payload.',
   123	            );
   124	        }
   125	
   126	        $ivLength = openssl_cipher_iv_length(
   127	            $this->cipher,
   128	        );
   129	
   130	        if ($ivLength === false) {
   131	            throw new RuntimeException(
   132	                sprintf(
   133	                    'Unable to determine IV length for cipher [%s].',
   134	                    $this->cipher,
   135	                ),
   136	            );
   137	        }
   138	
   139	        $minimumLength =
   140	            $ivLength
   141	            + self::TAG_LENGTH
   142	            + self::MINIMUM_PAYLOAD_LENGTH;

    94	        $directory = $this->getDirectory(
    95	            $baseDir,
    96	            $type,
    97	        );
    98	
    99	        if (!is_dir($directory)) {
   100	            return [];
   101	        }
   102	
   103	        $extension = $this->resolveExtension(
   104	            $type,
   105	        );
   106	
   107	        $pattern = $directory
   108	            . DIRECTORY_SEPARATOR
   109	            . '*.'
   110	            . $extension;

   113	
   114	        if ($files === false) {
   115	            throw new RuntimeException(

   120	            );

   133	            );

   137	            }
   138

   142	                path: $path,
   143	                extension: $extension,
   144	            );
   145	        }
   146

   142	    /**

   144	     *

   146	     * about the object represented by the envelope.
   147	     *
   148	     * @return array<string, mixed>
   149	     */
   150	    protected function decodeEnvelope(
   151	        EnvelopeContract $envelope,
   152	    ): array {
   153	        $codec = $this->resolveCodec($envelope);
   154	
   155	        $payload = $envelope->data;
   156	
   157	        /*
   158	         * Verify the signature against the exact bytes that
   159	         * are going to be unsealed.
   160	         */
   161	        if ($envelope->signed) {
   162	            $signer = $codec->signer
   163	                ?? throw new RuntimeException(
   164	                    'Envelope is signed but no signer is configured.',
   165	                );
   166	
   167	            if (!$signer->verify(
   168	                $payload,
   169	                $envelope->signature->value,
   170	            )) {
   171	                throw new RuntimeException(
   172	                    'Envelope signature verification failed.',
   173	                );
   174	            }
   175	        }
   176	
   177	        if ($envelope->sealed) {
   178	            $sealer = $codec->sealer
   179	                ?? throw new RuntimeException(
   180	                    'Envelope is sealed but no sealer is configured.',
   181	                );
   182	
   183	            $payload = $sealer->unseal(
   184	                $payload,
   185	            );
   186	        }
   187	
   188	        $decoded = $codec->decode(
   189	            $payload,
   190	        );
   191

     1	<?php
     2	
     3	declare(strict_types=1);
     4	
     5	namespace Codejitsu\Pipeline;
     6	
     7	use Closure;
     8	use RuntimeException;
     9	
    10	final class Pipeline
    11	{
    12	    /**
    13	     * @var list<callable>
    14	     */
    15	    private array $pipes = [];
    16	
    17	    /**
    18	     * @var mixed
    19	     */
    20	    private mixed $passable = null;
    21	
    22	    private ?Closure $destination = null;
    23	
    24	    public function send(
    25	        mixed $passable,
    26	        ?callable $destination = null,
    27	    ): mixed {
    28	        $this->passable = $passable;
    29	
    30	        if ($destination !== null) {
    31	            return $this->then($destination);
    32	        }
    33	
    34	        return $this;
    35	    }
    36	
    37	    public function through(
    38	        callable ...$pipes,
    39	    ): static {
    40	        foreach ($pipes as $pipe) {
    41	            $this->pipe($pipe);
    42	        }
    43	
    44	        return $this;
    45	    }
    46	
    47	    public function pipe(
    48	        callable $pipe,
    49	    ): static {
    50	        $this->pipes[] = $pipe;
    51	
    52	        return $this;
    53	    }
    54	
    55	    public function then(
    56	        callable $destination,
    57	    ): mixed {
    58	        $this->destination = Closure::fromCallable(
    59	            $destination,
    60	        );
    61	
    62	        $next = $this->destination;
    63	
    64	        foreach (
    65	            array_reverse($this->pipes)
    66	            as $pipe
    67	        ) {
    68	            $next = $this->wrap(
    69	                $pipe,
    70	                $next,
    71	            );
    72	        }
    73	
    74	        return $next($this->passable);
    75	    }
    76	
    77	    public function process(): mixed
    78	    {
    79	        if ($this->destination === null) {
    80	            throw new RuntimeException(
    81	                'Pipeline destination has not been configured.',
    82	            );
    83	        }
    84	
    85	        return $this->then(
    86	            $this->destination,
    87	        );
    88	    }
    89	
    90	    public function clear(): static
    91	    {
    92	        $this->pipes = [];
    93	        $this->passable = null;
    94	        $this->destination = null;
    95	
    96	        return $this;
    97	    }
    98	
    99	    private function wrap(
   100	        callable $pipe,
   101	        Closure $next,
   102	    ): Closure {
   103	        return static function (
   104	            mixed $passable,
   105	        ) use ($pipe, $next): mixed {
   106	            return $pipe(
   107	                $passable,
   108	                $next,
   109	            );
   110	        };
   111	    }
